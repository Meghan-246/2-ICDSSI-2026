import numpy as np


def triangle(row):
    conv = {'R': 0, 'G': 1, 'B': 2}
    inv = {0: 'R', 1: 'G', 2: 'B'}
    
    row = np.array([conv[c] for c in row], dtype=np.int64)
    n = len(row)
    
    if n == 1:
        return inv[int(row[0])]
    
    steps = n - 1
    
    # Descomponemos steps en base 3 y aplicamos cada potencia
    p = 1
    while p <= steps:
        p *= 3
    p //= 3
    
    remaining = steps
    while remaining > 0:
        p = 1
        while p * 3 <= remaining:
            p *= 3
        # Aplicar p pasos: combine es (2*(a+b)) % 3
        # Después de p=3^k pasos con esta operación:
        # coef = 2^p mod 3, y solo i=0 e i=p sobreviven
        coef = pow(2, p, 3)
        row = (coef * (row[:len(row)-p] + row[p:])) % 3
        remaining -= p
    
    return inv[int(row[0])]
